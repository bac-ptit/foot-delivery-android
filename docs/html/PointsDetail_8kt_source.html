122 private fun loadOrders() {
123 val userId = resolveUserIdForPoints()
124 RetrofitClient.apiService.getUserOrders(userId).enqueue(object : Callback<List<OrderResponse>> {
125 override fun onResponse(call: Call<List<OrderResponse>>, response: Response<List<OrderResponse>>) {
126 if (!response.isSuccessful || response.body().isNullOrEmpty()) {
127 Toast.makeText(this@PointsDetail, "Không có đơn hàng để đánh giá", Toast.LENGTH_SHORT).show()
128 return
129 }
130
131 val completedOrders = response.body()!!
132 .filter { it.status == "completed" }
133 .sortedByDescending { it.createdat }
134
135 firstOrderId = completedOrders.getOrNull(0)?.id
136 secondOrderId = completedOrders.getOrNull(1)?.id
137 thirdOrderId = completedOrders.getOrNull(2)?.id
138 fourthOrderId = completedOrders.getOrNull(3)?.id
139
140 bindOrderCard(R.id.tvDishName1, R.id.tvPoints1, completedOrders.getOrNull(0), "Đơn hàng #")
141 bindOrderCard(R.id.tvDishName2, R.id.tvPoints2, completedOrders.getOrNull(1), "Đơn hàng #")
142 bindOrderCard(R.id.tvDishName3, R.id.tvPoints3, completedOrders.getOrNull(2), "Đơn hàng #")
143 bindOrderCard(R.id.tvDishName4, R.id.tvPoints4, completedOrders.getOrNull(3), "Đơn hàng #")
144 }

184 private fun resolveUserIdForPoints(): Int {
185 val sharedPref = getSharedPreferences("user_prefs", Context.MODE_PRIVATE)
186 val userId = sharedPref.getInt("user_id", -1)
187 if (userId > 0) return userId
188
189 val userName = sharedPref.getString("user_name", "") ?: ""
190 if (userName.contains("Trung", ignoreCase = true) || userName.isBlank()) {
191 return 1
192 }
193 return 1
194 }
