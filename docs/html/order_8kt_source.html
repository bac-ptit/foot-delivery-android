286 private fun calculateDiscountAmount(total: Int): Int {
287 if (selectedDiscountValue <= 0) return 0
288 return if ((selectedDiscountType ?: "").equals("percent", ignoreCase = true)) {
289 ((total * selectedDiscountValue) / 100.0).toInt()
290 } else {
291 selectedDiscountValue
