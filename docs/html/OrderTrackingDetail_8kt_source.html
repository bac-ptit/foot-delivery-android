365 private fun formatOrderDate(dateString: String?): String {
366 if (dateString.isNullOrEmpty()) return "--"
367 return try {
368 // Expected format from backend: 2025-04-06T10:30:45
369 val parts = dateString.split("T")
370 if (parts.size >= 1) {
371 val dateParts = parts[0].split("-")
372 if (dateParts.size == 3) {
373 // Convert from 2025-04-06 to 06/04/2025
374 "${dateParts[2]}/${dateParts[1]}/${dateParts[0]}"
375 } else {
376 dateString
377 }
378 } else {
379 dateString
380 }
381 } catch (e: Exception) {
382 dateString
383 }
384 }
