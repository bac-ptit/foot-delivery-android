55async def login_for_access_token(form_data: OAuth2PasswordRequestForm = Depends(), db: Session = Depends(get_db)):
56 """Đăng nhập bằng username/password, trả về JWT access token."""
57 user = db.query(models.User).filter(models.User.username == form_data.username).first()
58 if not user or not auth.verify_password(form_data.password, user.password):
59 raise HTTPException(
60 status_code=status.HTTP_401_UNAUTHORIZED,
61 detail="Incorrect username or password",
62 headers={"WWW-Authenticate": "Bearer"},
63 )
64 access_token_expires = timedelta(minutes=auth.ACCESS_TOKEN_EXPIRE_MINUTES)
65 access_token = auth.create_access_token(
66 data={"sub": user.email}, expires_delta=access_token_expires
67 )
68 return {"access_token": access_token, "token_type": "bearer"}

319def build_menuitem_with_reviews(item: models.MenuItem, db: Session):
320 """Helper function to build MenuItem with reviews and avg_rating."""
321 restaurant = db.query(models.Restaurant).filter(models.Restaurant.id == item.restaurantid).first()
322 restaurant_name = restaurant.name if restaurant else ""
323
324 # Fetch reviews for this menu item
325 reviews = db.query(models.Review).filter(models.Review.menuitemid == item.id).all()
326 reviews_data = []
327 total_rating = 0
328
329 for review in reviews:
330 user = db.query(models.User).filter(models.User.id == review.userid).first()
331 user_name = user.name if user else "Unknown"
332 reviews_data.append({
333 "id": review.id,
334 "rating": review.rating,
335 "comment": review.comment,
336 "userid": review.userid,
337 "user_name": user_name
338 })
339 total_rating += review.rating
340
341 avg_rating = total_rating / len(reviews) if reviews else 0.0
342
343 return {
344 "id": item.id,
345 "name": item.name,
346 "image_url": item.image_url,
347 "price": item.price,
348 "is_available": item.is_available,
349 "description": item.description,
350 "restaurantid": item.restaurantid,
351 "categoryid": item.categoryid,
352 "restaurant_name": restaurant_name,
353 "reviews": reviews_data,
354 "avg_rating": avg_rating
355 }

422@app.get("/vnpay_return")
423async def vnpay_return(request: Request, db: Session = Depends(get_db)):
424 """Callback từ VNPay sau khi thanh toán, cập nhật trạng thái đơn hàng."""
425
426
427 params = dict(request.query_params)
428
429
430 response_code = params.get("vnp_ResponseCode")
431 order_ref = params.get("vnp_TxnRef") # Format: DH{order_id}
432
433 # Trích xuất order_id từ reference (VD: "DH123" -> 123)
434 try:
435 # Nếu là định dạng "DH{id}"
436 if order_ref and order_ref.startswith("DH"):
437 order_id = int(order_ref[2:])
438 else:
439 order_id = int(order_ref) if order_ref else 0
440 except (ValueError, AttributeError):
441 order_id = 0
442
443
444 if response_code == "00":
445 # Thanh toán thành công - cập nhật đơn hàng trong DB
446 if order_id > 0:
447 order = db.query(models.Orders).filter(models.Orders.id == order_id).first()
448 if order:
449 order.status = "paid"
450
451 # Tạo Payment record
452 payment_record = models.Payment(
453 status="success",
454 method="vnpay",
455 orderid=order.id
456 )
457 db.add(payment_record)
458 db.commit()
459 db.refresh(order)
460
461 # Gửi notification
462 total_formatted = f"{order.totalprice:,.0f}".replace(",", ".")
463 notify_user(
464 db=db,
465 user_id=order.userid,
466 title="Đơn hàng được đặt thành công! 🎉",
467 body=f"Đơn hàng #{order.id} đã được đặt thành công. Tổng tiền: {total_formatted}đ. Đang chờ quán xác nhận.",
468 order_id=order.id
469 )
470
471 return {
472 "status": "success",
473 "message": f"Thanh toán thành công cho đơn {order_ref}"
474 }
475 else:
476 # Thanh toán thất bại - cập nhật Payment record
477 if order_id > 0:
478 payment_record = models.Payment(
479 status="failed",
480 method="vnpay",
481 orderid=order_id
482 )
483 db.add(payment_record)
484 db.commit()
485
486 return {
487 "status": "failed",
488 "message": "Thanh toán thất bại"
489 }
