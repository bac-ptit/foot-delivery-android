102 override fun onResponse(call: Call<List<OrderResponse>>, response: Response<List<OrderResponse>>) {

170 private fun resolveUserIdForHistory(): Int {
171 val sharedPref = getSharedPreferences("user_prefs", Context.MODE_PRIVATE)
172 val userId = sharedPref.getInt("user_id", -1)
173 if (userId > 0) return userId
174
175 val userName = sharedPref.getString("user_name", "") ?: ""
176 if (userName.contains("Trung", ignoreCase = true) || userName.isBlank()) {
177 return 1
